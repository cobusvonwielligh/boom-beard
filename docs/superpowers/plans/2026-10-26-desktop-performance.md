**Goal:** Reduce desktop transfer size and improve LCP, repeat-load caching, accessibility, and SEO without changing layout or gameplay.

**Architecture:** Extend the existing Node static server with negotiated gzip and immutable caching for content-fingerprinted assets. Replace the two large PNG illustrations with same-dimension near-lossless WebP files, fingerprint the Lighthouse-flagged static assets, and add nonvisual HTML metadata and LCP priority.

**Tech Stack:** Node.js 20, `node:test`, built-in `node:zlib`, static HTML/CSS, WebP generated by `sharp-cli@6.1.0`

- [ ] **Step 1: Add raw-response compression tests**

In `test/cache.test.js`, import `node:http` and `gunzipSync` from `node:zlib`, then add this helper beside `base`:

```js
const rawGet = (url, headers = {}) => new Promise((resolve, reject) => {
  const req = http.get(base + url, { headers }, res => {
    const chunks = [];
    res.on('data', chunk => chunks.push(chunk));
    res.on('end', () => resolve({ status: res.statusCode, headers: res.headers, body: Buffer.concat(chunks) }));
  });
  req.on('error', reject);
});
```

Add these subtests inside `test('static caching', ...)` after the server starts:

```js
  await t.test('text responses negotiate gzip without changing their contents', async () => {
    const plain = await rawGet('/style.css');
    const zipped = await rawGet('/style.css', { 'Accept-Encoding': 'gzip' });
    assert.equal(plain.headers['content-encoding'], undefined);
    assert.equal(zipped.headers['content-encoding'], 'gzip');
    assert.equal(zipped.headers.vary, 'Accept-Encoding');
    assert.deepEqual(gunzipSync(zipped.body), plain.body);
    assert.ok(zipped.body.length < plain.body.length / 2,
      `gzip only reduced ${plain.body.length} bytes to ${zipped.body.length}`);
  });

  await t.test('range responses are never compressed', async () => {
    const ranged = await rawGet('/assets/audio/theme.mp3', {
      Range: 'bytes=0-1',
      'Accept-Encoding': 'gzip',
    });
    assert.equal(ranged.status, 206);
    assert.equal(ranged.headers['content-encoding'], undefined);
    assert.equal(ranged.body.length, 2);
  });
```

- [ ] **Step 3: Implement gzip negotiation in the static server**

Add the import:

```js
import { createGzip } from 'node:zlib';
```

Define the eligible extensions beside `MIME`:

```js
const COMPRESSIBLE = new Set(['.html', '.js', '.mjs', '.css', '.json', '.xml', '.txt']);
const MIN_GZIP_BYTES = 1024;
```

After `size` is known, calculate representation properties:

```js
  const compressible = COMPRESSIBLE.has(ext);
  const gzip = compressible
    && size >= MIN_GZIP_BYTES
    && !req.headers.range
    && /(?:^|,)\s*gzip\s*(?:,|$)/i.test(req.headers['accept-encoding'] || '');
```

Add `Vary` and `Content-Encoding` to `head`:

```js
    ...(compressible ? { Vary: 'Accept-Encoding' } : null),
    ...(gzip ? { 'Content-Encoding': 'gzip' } : null),
```

Replace the 304 response with variant-aware headers:

```js
  if (!req.headers.range && req.headers['if-none-match'] === etag) {
    res.writeHead(304, {
      'Cache-Control': head['Cache-Control'],
      ETag: etag,
      ...(head.Vary ? { Vary: head.Vary } : null),
      ...(head['Content-Encoding'] ? { 'Content-Encoding': head['Content-Encoding'] } : null),
    });
    return res.end();
  }
```

Before the ordinary uncompressed 200 response add:

```js
  if (gzip) {
    res.writeHead(200, head);
    return fs.createReadStream(file).pipe(createGzip()).pipe(res);
  }
```

- [ ] **Step 1: Write failing fingerprint, LCP, metadata, and asset tests**

Create `test/performance.test.js`:

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const read = rel => fs.readFileSync(path.join(ROOT, rel));
const text = rel => read(rel).toString('utf8');

const ASSETS = [
  'public/assets/img/mascot.8f66a578.webp',
  'public/assets/img/bomb-barrels.4639014b.webp',
  'public/assets/img/title-boom-beard-bandanas-bomb.1f959c46.webp',
  'public/assets/img/avatar.cf938d0b.png',
  'public/assets/fonts/clegan.7308d830.ttf',
];

function webpDimensions(rel) {
  const bytes = read(rel);
  assert.equal(bytes.subarray(0, 4).toString('ascii'), 'RIFF');
  assert.equal(bytes.subarray(8, 12).toString('ascii'), 'WEBP');
  let offset = 12;
  while (offset + 8 <= bytes.length) {
    const type = bytes.subarray(offset, offset + 4).toString('ascii');
    const length = bytes.readUInt32LE(offset + 4);
    const data = offset + 8;
    if (type === 'VP8L') {
      assert.equal(bytes[data], 0x2f, 'invalid VP8L signature');
      const bits = bytes.readUInt32LE(data + 1);
      return { width: (bits & 0x3fff) + 1, height: ((bits >>> 14) & 0x3fff) + 1 };
    }
    offset = data + length + (length & 1);
  }
  assert.fail(`${rel} has no VP8L dimensions`);
}

test('desktop metadata and the LCP image are explicit', () => {
  const html = text('public/index.html');
  assert.match(html, /<html lang="en">/);
  assert.match(html,
    /<meta\s+name="description"\s+content="Sixteen pirates\. One island\. Light the fuse and run\. Play Boom Beard, a fast browser arena game for up to 16 players\."\s*\/>/s);
  assert.match(html, /<link rel="canonical" href="https:\/\/boombeard\.co\.za\/" \/>/);
  assert.match(html,
    /src="\/assets\/img\/mascot\.8f66a578\.webp"[^>]*fetchpriority="high"/s);
  assert.doesNotMatch(html, /mascot\.png|bomb-barrels\.png|title-boom-beard-bandanas-bomb\.webp|avatar\.png|clegan\.ttf/);
});

test('all optimized asset references exist and old names are gone', () => {
  const css = text('public/style.css');
  const html = text('public/index.html');
  for (const rel of ASSETS) assert.ok(fs.existsSync(path.join(ROOT, rel)), `${rel} is missing`);
  assert.doesNotMatch(css,
    /mascot\.png|bomb-barrels\.png|title-boom-beard-bandanas-bomb\.webp|clegan\.ttf/);
  for (const rel of ASSETS) {
    const url = '/' + rel.replace(/^public\//, '');
    assert.ok(html.includes(url) || css.includes(url), `${url} is never referenced`);
  }
});

test('near-lossless WebPs preserve desktop dimensions and reduce bytes', () => {
  const mascot = 'public/assets/img/mascot.8f66a578.webp';
  const barrels = 'public/assets/img/bomb-barrels.4639014b.webp';
  assert.deepEqual(webpDimensions(mascot), { width: 384, height: 384 });
  assert.deepEqual(webpDimensions(barrels), { width: 440, height: 440 });
  assert.ok(read(mascot).length < 100_000, 'mascot WebP is unexpectedly large');
  assert.ok(read(barrels).length < 105_000, 'barrel WebP is unexpectedly large');
});

test('the server recognizes content-fingerprinted assets', () => {
  const server = text('server/index.js');
  assert.match(server, /fingerprinted.*max-age=31536000.*immutable/s);
});
```
